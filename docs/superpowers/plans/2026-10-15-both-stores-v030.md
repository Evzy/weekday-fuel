# Both Stores (v0.3.0) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Spec:** [`docs/superpowers/specs/2026-10-15-both-stores-design.md`](../specs/2026-10-15-both-stores-design.md). This plan covers release **v0.3.0** (catalogue, both stores, store switch, computed meal costs, snapshot prices). Live Woolworths prices (v0.3.1) get their own plan after this ships.

**Goal:** Every Shop item shows its real Coles (red) and Woolworths (green) product and price; Build a shop shows both totals and follows a Coles/Woolworths switch; meal costs are worked out from the chosen store's prices.

**Architecture:** A new pure module `catalogue.js` (like `cart.js`) holds every item's two products and all shopping maths (`shopList`, `storeTotals`, `mealCost`). Its functions are built by `create(items, amounts)` so tests use small fixtures; the real data is transcribed from an owner-approved research table. `Weekday_Fuel.html` only renders what the catalogue returns, and `cart.js` turns its Woolworths lines into trolley lines.

**Tech Stack:** plain browser JS (ES5 style in the asset modules), Node built-in test runner (`node --test`), GitHub Actions. No Kotlin changes.

## Global Constraints

- `applicationId` stays `dev.weekdayfuel.app`; the signing key does not change.
- Woolworths pages never get a JavaScript bridge into the app. Only origin `https://appassets.androidplatform.net` may message the host.
- Saved data keeps its shape except one new setting, `store: "coles" | "woolies"` (default `"woolies"`). Ticks stay keyed by item id.
- Colours: CSS variables `--coles` and `--woolies`. Light theme `--coles:#C8102E; --woolies:#1A7F3C;`, both dark blocks `--coles:#FF7A7A; --woolies:#5FC58D;`.
- Store names shown to the owner: "Coles" and "Woolworths". Internal keys: `coles`, `woolies`.
- Ham: `deliGrams(need) = max(100, ceil(need / 100) * 100)` grams at both stores; trolley quantity in kilograms.
- Chicken small pack: used while need ≤ small pack size + 50 g.
- Meal Light/Extra cost = computed Standard × (typed L or X ÷ typed S).
- The trolley stays Woolworths-only.
- Nothing in CI talks to the real Coles or Woolworths sites.
- Commit straight to `main`, prefix `both-stores:` (release commit `release: v0.3.0`). After each push, CI must be green before the next task.
- No local Android SDK. JS tests run locally with Node 24: `node --test "apps/android/app/src/test/js/*.test.js"` (quote the glob; a bare directory fails on Windows).
- Release (version bump) only in Task 6.
- Do not touch `platra/`, the keystore, `local.properties`, or `docs/seed/`.

```js
// Item (catalogue.js ITEMS)
{ id, sec /* 0 fixtures, 1 weekly top-up, 2 pantry & freezer */, name, unit /* "g", "mL", or a count word: "eggs", "wraps", "slices", "pouches", "bottles", "bars", "cucumbers", "capsicums", "muffins" */,
  note, same /* identical product at both stores */,
  coles: Product | null, woolies: Product | null }

{ name, url, size /* one pack, in the item's unit; 1 for per-kg */, price /* $ per pack, or per kg */, checked /* "YYYY-MM-DD" */,

// Line: { product /* the pack actually bought */, quantity /* packs, or kg for perKg */, cost, buy /* text */ }
```

```js
/* Coles and Woolworths products for every Shop item, and the shopping maths. Pure: no DOM, no storage. */
(function (root) {
  "use strict";
  /* A small pack is used while the need is at most this much over its size (in the item's unit). */
  var SMALL_PACK_SLACK = 50;

  /* Filled from docs/catalogue-research.md in Task 3. */
  var ITEMS = [];
  var AMOUNTS = {};

  function round2(x) {
    return Math.round(x * 100) / 100;
  }

  /* Deli counters sell by 100 g, minimum 100 g. Round up so the owner never runs short. */
  function deliGrams(needGrams) {
    return Math.max(100, Math.ceil(needGrams / 100) * 100);
  }

  function packLabel(product, unit) {
    return product.size + (/^(g|mL)$/.test(unit) ? unit : " " + unit);
  }

  function storeLine(item, product, need) {
    if (product.perKg) {
      var grams = deliGrams(need);
      return { product: product, quantity: grams / 1000, cost: round2((grams / 1000) * product.price), buy: grams + "g from the deli" };
    }
    var chosen = product.small && need <= product.small.size + SMALL_PACK_SLACK ? product.small : product;
    var packs = Math.max(1, Math.ceil(need / chosen.size - 1e-9));
    return {
      product: chosen,
      quantity: packs,
      cost: round2(packs * chosen.price),
      buy: chosen.size === 1 ? packs + " " + item.unit : packs + " × " + packLabel(chosen, item.unit),
    };
  }

  function create(items, amounts) {
    var byId = {};
    items.forEach(function (it) {
      byId[it.id] = it;
    });

    function shopList(picks, fixtureDays) {
      var need = {};
      var used = {};
      Object.keys(picks).forEach(function (meal) {
        var serves = picks[meal];
        var per = amounts[meal];
        if (!serves || !per) return;
        Object.keys(per).forEach(function (id) {
          need[id] = (need[id] || 0) + per[id] * serves;
          (used[id] = used[id] || []).push(meal);
        });
      });
      if (fixtureDays > 0) {
        items.forEach(function (it) {
          if (it.sec === 0) need[it.id] = fixtureDays;
        });
      }
      return items
        .filter(function (it) {
          return need[it.id] > 0;
        })
        .map(function (it) {
          var n = need[it.id];
          return {
            id: it.id,
            sec: it.sec,
            name: it.name,
            unit: it.unit,
            need: n,
            used: used[it.id] || [],
            coles: it.coles ? storeLine(it, it.coles, n) : null,
            woolies: it.woolies ? storeLine(it, it.woolies, n) : null,
          };
        });
    }

    function storeTotals(rows, store, ticks) {
      var t = { total: 0, left: 0, missed: [] };
      rows.forEach(function (r) {
        var line = r[store];
        if (!line) {
          t.missed.push(r.id);
          return;
        }
        t.total += line.cost;
        if (!ticks[r.id]) t.left += line.cost;
      });
      t.total = round2(t.total);
      t.left = round2(t.left);
      return t;
    }

    function mealCost(meal, store) {
      var per = amounts[meal];
      if (!per) return null;
      var other = store === "coles" ? "woolies" : "coles";
      var sum = 0;
      Object.keys(per).forEach(function (id) {
        var it = byId[id];
        var p = it && (it[store] || it[other]);
        if (!p) return;
        sum += p.perKg ? (per[id] / 1000) * p.price : (per[id] / p.size) * p.price;
      });
      return round2(sum);
    }

    function usedIn(id) {
      return Object.keys(amounts).filter(function (meal) {
        return id in amounts[meal];
      });
    }

    function colesChecked() {
      var dates = [];
      items.forEach(function (it) {
        if (!it.coles) return;
        dates.push(it.coles.checked);
        if (it.coles.small) dates.push(it.coles.small.checked);
      });
      return dates.sort()[0] || null;
    }

    return {
      ITEMS: items,
      AMOUNTS: amounts,
      item: function (id) {
        return byId[id];
      },
      deliGrams: deliGrams,
      shopList: shopList,
      storeTotals: storeTotals,
      mealCost: mealCost,
      usedIn: usedIn,
      colesChecked: colesChecked,
    };
  }

  var api = create(ITEMS, AMOUNTS);
  api.create = create;
  if (typeof module === "object" && module.exports) module.exports = api;
  else root.WeekdayFuelCatalogue = api;
})(this);
```

```js
const MEALS = ["toasty", "eggmuffin", "yogbowl", "brekwrap", "tenderwrap", "burritobowl", "teriyaki", "buffalomac", "schnitzel", "koreanbeef", "bolognese", "redcurry", "friedrice", "snackplate"];

const COLES = /^https:\/\/www\.coles\.com\.au\/product\//;
const DATE = /^\d{4}-\d{2}-\d{2}$/;

function checkProduct(where, p, store) {
  assert.ok(p.name, where + " name");
  assert.match(p.url, store === "coles" ? COLES : WOOLIES, where + " url");
  assert.ok(p.price > 0, where + " price");
  assert.match(p.checked, DATE, where + " checked");
  assert.ok(p.perKg ? p.size === 1 : p.size > 0, where + " size");
  if (store === "woolies") assert.equal(p.stockcode, Number(WOOLIES.exec(p.url)[1]), where + " stockcode matches url");

test("every item is well formed", () => {
  const ids = new Set();
  for (const it of cat.ITEMS) {
    assert.ok(!ids.has(it.id), "duplicate " + it.id);
    ids.add(it.id);
    assert.ok([0, 1, 2].includes(it.sec), it.id + " sec");
    assert.ok(it.name && it.unit, it.id + " name/unit");
    assert.ok(it.coles || it.woolies, it.id + " has at least one store");
    if (it.same) assert.ok(it.coles && it.woolies, it.id + " same needs both");
    for (const store of ["coles", "woolies"]) {
      const p = it[store];

  assert.deepEqual(Object.keys(cat.AMOUNTS).sort(), [...MEALS].sort());
```

- [ ] **Step 2: Rewrite `cart.js`**

```js
/* Turns Build a shop rows into Woolworths trolley lines. Pure: no DOM, no storage. */
(function (root) {
  "use strict";

  function cartLines(rows, ticks) {
    var lines = [];
    var manual = [];
    rows.forEach(function (row) {
      if (ticks[row.id]) return;
      var line = row.woolies;
      if (!line) {
        manual.push({ itemId: row.id, name: row.coles ? row.coles.product.name : row.name, reason: row.coles ? "coles" : "nostore" });
        return;
      }
      var p = line.product;
      var quantity = Math.round(line.quantity * 1000) / 1000;
      lines.push({
        itemId: row.id,
        stockcode: p.stockcode,
        quantity: quantity,
        name: p.perKg ? p.name + " (" + Math.round(quantity * 1000) + "g)" : p.name,
        url: p.url,
      });
    });
    return { lines: lines, manual: manual };
  }

  var api = { cartLines: cartLines };
  if (typeof module === "object" && module.exports) module.exports = api;
  else root.WeekdayFuelCart = api;
})(this);
```
