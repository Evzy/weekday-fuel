# Plans Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Plans panel on the Meals tab with one fixed week ("Fewest items"). It fills empty breakfast, lunch and dinner slots on 5 weekdays from a chosen start day, fits portions and marks shop days. The starter shop is removed.

**Architecture:** A new pure module `plans.js` (`window.WeekdayFuelPlans`) holds the plan data, the date and fill logic, and the summary maths. `Weekday_Fuel.html` renders the panel and sheet, calls `fill`, then fits portions with the existing `fitTiers` and saves. Spec: `docs/superpowers/specs/2026-10-19-plans-design.md`.

**Tech Stack:** Vanilla JS (ES5 in modules, ES2017 in HTML), node:test.

## Global Constraints

- Commit straight to `main`, staging by explicit path. Never stage `.superpowers/`. Never `git stash`, `git clean` or `git reset --hard`.
- Escape every meal-, catalogue- and plan-derived string in HTML with `esc`.
- Existing localStorage keys keep their shape; additive optional fields are OK. Meal ids never change.
- Meal objects on days are `{id, slot, t}`, with `t` one of `"L"`, `"S"` or `"X"`. Slots are `breakfast`, `lunch`, `dinner` and `snack`.
- Bump the version only in Task 3.
- Run all tests with `node --test "apps/android/app/src/test/js/*.test.js"`.
- Parse check: extract the last inline `<script>` of `Weekday_Fuel.html` to `%TEMP%/wf.js` and run `node --check` on it.

```js
/* Plans: fixed preset weeks. Pure: no DOM, no storage. */
(function (root) {
  "use strict";

  var SLOTS = ["breakfast", "lunch", "dinner"];
  var ORDER = ["breakfast", "lunch", "snack", "dinner"];

  var PLANS = [
    {
      id: "fewest",
      name: "Fewest items",
      blurb: "4 meals that share their ingredients, so the shopping list is as short as it gets.",
      days: [
        { breakfast: "toasty", lunch: "tenderwrap", dinner: "buffalomac" },
        { breakfast: "toasty", lunch: "burritobowl", dinner: "buffalomac" },
        { breakfast: "toasty", lunch: "tenderwrap", dinner: "burritobowl" },
        { breakfast: "toasty", lunch: "burritobowl", dinner: "buffalomac" },
        { breakfast: "toasty", lunch: "tenderwrap", dinner: "burritobowl" },
      ],
    },
  ];

  function byId(id) {
    for (var i = 0; i < PLANS.length; i++) if (PLANS[i].id === id) return PLANS[i];
    return null;
  }

  function parse(iso) {
    var p = String(iso).split("-");
    return new Date(Date.UTC(+p[0], +p[1] - 1, +p[2]));
  }

  function fmt(d) {
    return d.toISOString().slice(0, 10);
  }

  function isWeekend(iso) {
    var w = parse(iso).getUTCDay();
    return w === 0 || w === 6;
  }

  function weekdaysFrom(startIso, n) {
    var out = [];
    var d = parse(startIso);
    while (out.length < n) {
      var iso = fmt(d);
      if (!isWeekend(iso)) out.push(iso);
      d.setUTCDate(d.getUTCDate() + 1);
    }
    return out;
  }

  function mealsOf(daysMap, iso) {
    var day = daysMap && daysMap[iso];
    return day && Array.isArray(day.meals) ? day.meals : [];
  }

  function hasSlot(meals, slot) {
    return meals.some(function (m) {
      return m && m.slot === slot;
    });
  }

  function defaultStart(todayIso, daysMap) {
    var days = weekdaysFrom(todayIso, 10);
    for (var i = 0; i < days.length; i++) {
      var meals = mealsOf(daysMap, days[i]);
      if (!SLOTS.some(function (sl) { return hasSlot(meals, sl); })) return days[i];
    }
    return days[0];
  }

  function fill(plan, startIso, daysMap) {
    var dates = weekdaysFrom(startIso, plan.days.length);
    var changes = {};
    var added = 0;
    var kept = 0;
    dates.forEach(function (iso, i) {
      var existing = mealsOf(daysMap, iso);
      var meals = existing.slice();
      var filled = [];
      SLOTS.forEach(function (sl) {
        if (hasSlot(existing, sl)) {
          kept += 1;
          return;
        }
        meals.push({ id: plan.days[i][sl], slot: sl, t: "S" });
        filled.push(sl);
        added += 1;
      });
      if (!filled.length) return;
      meals = meals
        .map(function (m, idx) { return { m: m, idx: idx }; })
        .sort(function (a, b) {
          var d = ORDER.indexOf(a.m.slot) - ORDER.indexOf(b.m.slot);
          return d || a.idx - b.idx;
        })
        .map(function (x) { return x.m; });
      changes[iso] = { meals: meals, added: filled };
    });
    return { dates: dates, changes: changes, added: added, kept: kept };
  }

  function mealCounts(plan) {
    var out = {};
    plan.days.forEach(function (d) {
      SLOTS.forEach(function (sl) {
        out[d[sl]] = (out[d[sl]] || 0) + 1;
      });
    });
    return out;
  }

  function summary(plan, shopFn) {
    var r = shopFn(mealCounts(plan)) || { items: 0, total: 0 };
    return { items: r.items, costPerDay: Math.round((r.total / plan.days.length) * 100) / 100 };
  }

  var api = {
    PLANS: PLANS,
    byId: byId,
    isWeekend: isWeekend,
    weekdaysFrom: weekdaysFrom,
    defaultStart: defaultStart,
    fill: fill,
    mealCounts: mealCounts,
    summary: summary,
  };
  if (typeof module === "object" && module.exports) module.exports = api;
  else root.WeekdayFuelPlans = api;
})(this);
```

Note: `defaultStart` searches 10 weekdays, which is 14 calendar days. `meals.push` reuses the existing meal objects for kept meals, so the bought test's `deepEqual` holds, and the input day's array is never modified because `slice()` copies it.
