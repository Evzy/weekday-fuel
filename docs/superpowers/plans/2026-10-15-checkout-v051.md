# Checkout part 2 (v0.5.1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Spec:** [`docs/superpowers/specs/2026-10-15-checkout-part2-design.md`](../specs/2026-10-15-checkout-part2-design.md)

**Goal:** The in-app Checkout panel also sets substitutions, the driver note and leave unattended, and the delivery address, so Woolworths' own checkout only needs Place order.

**Architecture:** `assets/woolworths/checkout.js` gains three write requests (`subs`, `options`, `address`) and two load sections (`options`, `addresses`) plus `summary.items`. The Kotlin host validates three new page messages (`checkoutSubs`, `checkoutOptions`, `checkoutAddress`) and runs them like `checkoutBook`; every non-load request is a write that is never re-sent. `checkoutview.js` gains pure helpers for the substitution list and address rows, and `Weekday_Fuel.html` draws three new parts and generalises "booking" into one "saving" state.

**Tech Stack:** Plain JS (ES5-style IIFEs with `async`, Node `node:test`), one-file HTML app, Kotlin + Android WebView, `org.json`, GitHub Actions CI.

## Global Constraints

- The app never places the order or pays. No pick-up in the app: a pick-up order (`DeliveryMethod` not `"Courier"`) gets no address, options or substitutions sections, and every write on it answers "not a delivery order" without sending anything.
- Substitutions: one switch per trolley item; **Allow all**, **None**; **Save** sends only the items whose switch differs from Woolworths, in one request. Items with `isexcludedfromsubstitution` `"True"` show "No substitutes" and no switch.
- Driver note at most 500 characters; **Leave unattended** only when `ShouldShowCanLeaveUnattendedPreference` is not `false`. The pick-up note is sent back unchanged.
- Address: only the owner's saved addresses; choosing one switches straight away; hint "Switching may clear your delivery time." No adding or editing addresses.
- Boosts run only on load (open, Refresh, Try again), never on a write.
- One request at a time: while a load or write runs, every Save, Book, address choice, switch, View on Woolworths, Pay and Refresh is disabled; replies for another request are ignored.
- A write whose reply is lost says "Couldn't confirm the change with Woolworths. Tap Refresh to check." Nothing is re-sent.
- Woolworths pages never get a JavaScript bridge. Only origin `https://appassets.androidplatform.net` may message the host. Every Woolworths-derived string in the page goes through `esc`, attributes included. Nothing in CI talks to the real Coles or Woolworths sites.
- Commit straight to `main` with the prefix `checkout:`; the release commit is `release: v0.5.1`. CI green before the next task. No local Android SDK: Kotlin is compiled and tested only by CI.
- Bump the version only in Task 5. Do not touch `platra/`, the keystore, `local.properties`, `docs/seed/`, or the untracked `.superpowers/` files.

```js
  function deliveryOptions(o) {
    if (!o || typeof o !== "object") throw new Error("options response has no options");
    return {
      instructions: typeof o.DeliveryInstructions === "string" ? o.DeliveryInstructions : "",
      leaveUnattended: o.CanLeaveUnattended === true,
      offerUnattended: o.ShouldShowCanLeaveUnattendedPreference !== false,
    };
  }

  async function readOptions() {
    return deliveryOptions(await getJson(URLS.options));
  }

  async function readAddresses(dlv) {
    var a = await getJson(URLS.addresses);
    if (!a || !Array.isArray(a.Address)) throw new Error("addresses response has no list");
    return {
      current: dlv.address.addressId || null,
      list: a.Address.filter(Boolean)
        .map(function (x) {
          return { id: x.AddressId, text: typeof x.AddressText === "string" ? x.AddressText.replace(/\s+/g, " ").trim() : "", primary: x.IsPrimary === true };
        })
        .filter(function (x) { return typeof x.id === "number" && x.id > 0; }),
    };
  }

  async function saveSubs(req) {
    try {
      var j = await postJson(URLS.subs, {
        Items: req.items.map(function (i) { return { Stockcode: i.stockcode, AllowSubstitution: i.allow }; }),
      });
      return j && j.Success === true ? null : (j && j.ErrorMessage) || "not saved";
    } catch (e) {
      return message(e);
    }
  }

  /* The pick-up note is read first and sent back as it is. */
  async function saveOptions(req) {
    try {
      var current = await getJson(URLS.options);
      var saved = await postJson(URLS.options, {
        CanLeaveUnattended: req.leaveUnattended,
        PickupInstructions: current && typeof current.PickupInstructions === "string" ? current.PickupInstructions : "",
        DeliveryInstructions: req.instructions,
      });
      var o = deliveryOptions(saved);
      o.offerUnattended = !current || current.ShouldShowCanLeaveUnattendedPreference !== false;
      return { error: null, options: o };
    } catch (e) {
      return { error: message(e), options: null };
    }
  }

  async function switchAddress(req) {
    try {
      var a = await getJson(URLS.addresses);
      var known = a && Array.isArray(a.Address) && a.Address.some(function (x) { return x && x.AddressId === req.addressId; });
      if (!known) return "not one of your saved addresses";
      var j = await postJson(URLS.book, { fulfilmentMethod: "Courier", addressId: req.addressId });
      return j && j.IsSuccessful === true ? null : (j && j.Message) || "not switched";
    } catch (e) {
      return message(e);
    }
  }
```

```js
  function validRequest(req) {
    if (!req) return false;
    switch (req.action) {
      case "load":
        return true;
      case "book":
        return Number.isFinite(req.timeslotId) && req.timeslotId > 0 && /^\d{4}-\d{2}-\d{2}$/.test(String(req.windowDate));
```

```kotlin
    /** The page's `checkoutSubs` message as the script's subs request, or null unless it has 1–200 items, each a positive whole stockcode and a boolean. */
    fun subsRequest(msg: JSONObject): String? {
        val items = msg.optJSONArray("items") ?: return null
        if (items.length() !in 1..MAX_SUBS) return null
        val out = JSONArray()
        for (i in 0 until items.length()) {
            val item = items.optJSONObject(i) ?: return null
            val code = wholeNumber(item.opt("stockcode")) ?: return null
            val allow = item.opt("allow") as? Boolean ?: return null
            out.put(JSONObject().put("stockcode", code).put("allow", allow))
        }
        return JSONObject().put("action", "subs").put("items", out).toString()
    }

    /** The page's `checkoutOptions` message as the script's options request, or null unless the note is text of at most 500 characters and the switch a boolean. */
    fun optionsRequest(msg: JSONObject): String? {
        val note = msg.opt("instructions") as? String ?: return null
        val unattended = msg.opt("leaveUnattended") as? Boolean ?: return null
        if (note.length > MAX_NOTE) return null
        return JSONObject().put("action", "options").put("instructions", note).put("leaveUnattended", unattended).toString()
    }

    /** The page's `checkoutAddress` message as the script's address request, or null unless the id is a positive whole number. */
    fun addressRequest(msg: JSONObject): String? {
        val id = wholeNumber(msg.opt("addressId")) ?: return null
        return JSONObject().put("action", "address").put("addressId", id).toString()
    }

    /** A positive whole number JavaScript holds exactly; strings and fractions are refused. */
    private fun wholeNumber(value: Any?): Long? {
        val n =
            when (value) {
                is Int -> value.toLong()
                is Long -> value
                else -> return null
            }
        return n.takeIf { it in 1..MAX_SAFE_INTEGER }
    }
```
